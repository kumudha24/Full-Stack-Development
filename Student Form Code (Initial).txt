✔️server.js
const express = require("express");
const cors = require("cors");
const path = require("path");
const db = require("./db");

const app = express();

app.use(cors());
app.use(express.json());

// ✅ FIXED STATIC PATH
app.use(express.static(path.join(__dirname, "../public")));

/* INSERT */
app.post("/add", (req, res) => {
  const { name, email, age } = req.body;
  const sql = "INSERT INTO students (name, email, age) VALUES (?, ?, ?)";
  db.query(sql, [name, email, age], () => {
    res.send("Inserted");
  });
});

/* FETCH */
app.get("/students", (req, res) => {
  db.query("SELECT * FROM students", (err, result) => {
    res.send(result);
  });
});

/* DELETE */
app.delete("/delete/:id", (req, res) => {
  db.query("DELETE FROM students WHERE id=?", [req.params.id], () => {
    res.send("Deleted");
  });
});

app.listen(3000, () => {
  console.log("Server running on port 3000");
});

✔️script.js
function addStudent() {
  fetch("http://localhost:3000/add", {
    method: "POST",
    headers: { "Content-Type": "application/json" },
    body: JSON.stringify({
      name: document.getElementById("name").value,
      email: document.getElementById("email").value,
      age: document.getElementById("age").value
    })
  }).then(() => {
    loadStudents();
  });
}

function loadStudents() {
  fetch("http://localhost:3000/students")
    .then(res => res.json())
    .then(data => {
      let rows = "";
      data.forEach(s => {
        rows += `
          <tr>
            <td>${s.id}</td>
            <td>${s.name}</td>
            <td>${s.email}</td>
            <td>${s.age}</td>
            <td>
              <button class="delete" onclick="deleteStudent(${s.id})">Delete</button>
            </td>
          </tr>`;
      });
      document.getElementById("tableData").innerHTML = rows;
    });
}

function deleteStudent(id) {
  fetch(`http://localhost:3000/delete/${id}`, {
    method: "DELETE"
  }).then(() => loadStudents());
}

loadStudents();
